cmake_minimum_required(VERSION 3.25)

set(FIRMWARE_IDENTITY_FILE ${CMAKE_CURRENT_SOURCE_DIR}/identity.json)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${FIRMWARE_IDENTITY_FILE})
file(READ ${FIRMWARE_IDENTITY_FILE} FIRMWARE_IDENTITY)
set(FIRMWARE_DEFINITIONS "")
foreach(field name short_name id version data_directory update_file playlist_start_directive)
    string(TOUPPER ${field} variable)
    string(JSON FIRMWARE_${variable} GET "${FIRMWARE_IDENTITY}" ${field})
    list(APPEND FIRMWARE_DEFINITIONS FIRMWARE_${variable}="${FIRMWARE_${variable}}")
endforeach()
string(REGEX MATCH "^[0-9]+\\.[0-9]+" FIRMWARE_REVISION "${FIRMWARE_VERSION}")
if(NOT FIRMWARE_REVISION)
    message(FATAL_ERROR "identity.json: version \"${FIRMWARE_VERSION}\" does not start with major.minor")
endif()
list(APPEND FIRMWARE_DEFINITIONS FIRMWARE_REVISION="${FIRMWARE_REVISION}")

project(${FIRMWARE_SHORT_NAME} VERSION ${FIRMWARE_VERSION} LANGUAGES C)

if(WIN32)
    enable_language(RC)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(ECHO_SANITIZE "Build host code with UndefinedBehaviorSanitizer" OFF)
option(ECHO_ADDRESS_SANITIZER "Build host code with AddressSanitizer" OFF)
option(ECHO_BUILD_SIM "Build the SDL2 simulator" ON)
option(ECHO_BUILD_TESTS "Build the host tests" ON)
option(ECHO_BUILD_DEVICE "Build the device firmware with cmake/arm-none-eabi.cmake" OFF)
option(ECHO_RELEASE "Leave the developer-only Diagnostics rows out of the build" OFF)
option(ECHO_BOOT_TRACE "Show each start-up step on screen and skip the updater" OFF)
set(ECHO_PANEL_MADCTL 0x28 CACHE STRING "Memory access control byte of the full-size Echo panel (0x28, or 0x68 to mirror columns)")
set(ECHO_APP_RAM_BUDGET_BYTES 131072 CACHE STRING "Static RAM budget for src/ on the device")

include(cmake/project_options.cmake)

add_subdirectory(src)
add_subdirectory(platform/device/logic)

if(ECHO_BUILD_SIM)
    add_subdirectory(platform/sim)
endif()

if(ECHO_BUILD_DEVICE)
    add_subdirectory(platform/device)
endif()

if(ECHO_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
